def preprocess_text(text): 
    if not isinstance(text, str): 
        return "" 
    text = text.lower() 
    text = text.translate(str.maketrans('', '', 
string.punctuation)) 
    tokens = word_tokenize(text) 
    stop_words = set(stopwords.words('english')) 
    tokens = [word for word in tokens if word not in stop_words] 
    lemmatizer = WordNetLemmatizer() 
    tokens = [lemmatizer.lemmatize(word) for word in tokens] 
    preprocessed_text = ' '.join(tokens) 
    return preprocessed_text

def extract_topics(texts, n_topics=5): 
    if not texts or len(texts) == 0: 
        return [], [] 
    try: 
        vectorizer = CountVectorizer(max_df=0.95, min_df=2, 
stop_words='english') 
        tf = vectorizer.fit_transform(texts) 
        if tf.shape[0] > 0: 
            lda = 
LatentDirichletAllocation(n_components=n_topics, 
random_state=42) 
            lda.fit(tf) 
            feature_names = vectorizer.get_feature_names_out() 
            topics = [] 
            for topic_idx, topic in enumerate(lda.components_): 
                top_features_ind = topic.argsort()[-10:]  # Top 
10 words per topic 
                top_features = [feature_names[i] for i in 
top_features_ind] 
                topics.append(" ".join(top_features)) 
            topic_labels = lda.transform(tf).argmax(axis=1) 
            return topics, topic_labels 
    except Exception as e: 
        app.logger.error(f"Error in topic extraction: {e}") 
        return [], []

def analyze_text_for_actions(text): 
    actions = [] 
    text_lower = text.lower() 
    # Define actionable patterns 
    action_triggers = { 
        'response time': ['slow response', 'take too long', 
'didn\'t respond'], 
        'bug': ['bug', 'error', 'crash', 'not working'], 
        'feature request': ['wish', 'would like', 'please add', 
'need feature'], 
        'pricing': ['too expensive', 'cost too much', 
'overpriced', 'price high'], 
        'documentation': ['documentation poor', 'no manual', 
'hard to understand', 'lack of guidance'], 
        'ui/ux': ['hard to use', 'confusing interface', 'bad 
design', 'poor layout'] 
    } 
    for category, patterns in action_triggers.items(): 
        for pattern in patterns: 
            if pattern in text_lower: 
                actions.append((category, pattern)) 
                break  # Only add each category once 
    return actions 
def generate_insights(df): 
    if df.empty: 
        return {}, df 
    try: 
        # Sentiment Analysis 
        df[['sentiment', 'sentiment_score']] = df['text'].apply( 
            lambda x: pd.Series(analyze_sentiment(x)) 
        ) 
        # Topic Extraction 
        topics, topic_labels = 
extract_topics(df['text'].tolist()) 
        if topics: 
            df['topic'] = topic_labels 
            df['topic_label'] = df['topic'].apply(lambda x: 
topics[x] if x < len(topics) else "unknown") 
        else: 
            df['topic_label'] = "unknown" 
        # Key phrase extraction 
        df['key_phrases'] = df['text'].apply(get_key_phrases) 
        # Actionable insights 
        df['actionable_items'] = 
df['text'].apply(analyze_text_for_actions) 
        # Convert date strings to datetime objects for trend 
analysis 
        df['date'] = pd.to_datetime(df['date'], errors='coerce') 
        # Generate comprehensive insights 
        insights = { 
            'sentiment_distribution': 
dict(df['sentiment'].value_counts()), 
            'topic_distribution': 
dict(df['topic_label'].value_counts()) if 'topic_label' in 
df.columns else {}, 
            'top_positive_feedback': df[df['sentiment'] == 
'positive']['text'].tolist()[:5], 
            'top_negative_feedback': df[df['sentiment'] == 
'negative']['text'].tolist()[:5], 
            'common_key_phrases': list(Counter([phrase for 
sublist in df['key_phrases'].tolist() for phrase in 
sublist]).most_common(10)), 
            'actionable_items': list(Counter([item[0] for 
sublist in df['actionable_items'].tolist() for item in 
sublist]).most_common(10)), 
        } 
        # Sentiment trends over time 
        if 'date' in df.columns and not 
df['date'].isnull().all(): 
            df['month'] = df['date'].dt.to_period('M') 
            sentiment_trends = df.groupby(['month', 
'sentiment']).size().unstack(fill_value=0) 
            if not sentiment_trends.empty: 
                insights['sentiment_over_time'] = [{ 
                    'date': str(month), 
                    'positive': int(row.get('positive', 0)), 
                    'negative': int(row.get('negative', 0)), 
                    'neutral': int(row.get('neutral', 0)) 
                } for month, row in 
sentiment_trends.reset_index().iterrows()] 
        # Topic trends over time 
        if 'date' in df.columns and not 
df['date'].isnull().all() and 'topic_label' in df.columns: 
            topic_trends = df.groupby(['month', 
'topic_label']).size().unstack(fill_value=0) 
            if not topic_trends.empty: 
                # Convert to list of dicts for easier handling 
in JS 
                topic_trends_dict = {} 
                for month in topic_trends.index: 
                    month_str = str(month) 
                    topic_trends_dict[month_str] = 
topic_trends.loc[month].to_dict() 
                insights['topic_trends'] = topic_trends_dict 
        # Top actionable items with examples 
        actionable_counts = Counter() 
        actionable_examples = {} 
        for idx, row in df.iterrows(): 
            for category, pattern in row['actionable_items']: 
                actionable_counts[category] += 1 
                if category not in actionable_examples: 
                    actionable_examples[category] = [] 
                
actionable_examples[category].append(row['text']) 
        # Sort by count and pick top 5 
        top_actionable_items = actionable_counts.most_common(5) 
        insights['top_actionable_items'] = [ 
            { 
                'category': category, 
                'count': count, 
                'examples': actionable_examples.get(category, 
[])[:3]  # Top 3 examples 
            } 
            for category, count in top_actionable_items 
        ] 
        return insights, df 
    except Exception as e: 
        app.logger.error(f"Error in generating insights: {e}") 
        return {}, df 
class FeedbackDataset(Dataset): 
    def __init__(self, texts, labels, tokenizer, 
max_length=512): 
        self.texts = texts 
        self.labels = labels 
        self.tokenizer = tokenizer 
        self.max_length = max_length 
    def __len__(self): 
        return len(self.texts) 
    def __getitem__(self, idx): 
        text = str(self.texts[idx]) 
        label = self.labels[idx] 
        encoding = self.tokenizer( 
            text, 
            max_length=self.max_length, 
            padding='max_length', 
            truncation=True, 
            return_tensors='pt' 
        ) 
        return { 
            'input_ids': encoding['input_ids'].flatten(), 
            'attention_mask': 
encoding['attention_mask'].flatten(), 
            'labels': torch.tensor(label, dtype=torch.long) 
        }

class FeedbackDataset(Dataset):
def __init__(self, texts, labels, tokenizer,
max_length=512):
self.texts = texts
self.labels = labels
self.tokenizer = tokenizer
self.max_length = max_length
def __len__(self):
return len(self.texts)
def __getitem__(self, idx):
text = str(self.texts[idx])
label = self.labels[idx]
encoding = self.tokenizer(
text,
max_length=self.max_length,
padding='max_length',
truncation=True,
return_tensors='pt'
)
return {
'input_ids': encoding['input_ids'].flatten(),
'attention_mask':
encoding['attention_mask'].flatten(),
'labels': torch.tensor(label, dtype=torch.long)
}
