@dataclass
class TemporalPattern:
    """Represents a discovered temporal pattern"""
    pattern_id: str
    time_scale: float  # In seconds
    pattern_type: str
    strength: float

    impact: float

class MultiScaleTemporalAnalyzer:
    """Analyzes patterns across multiple time scales simultaneously"""
    
    def __init__(self, scales: np.ndarray):
        self.scales = scales  # From microseconds to years
        self.wavelet = 'morlet'  # Best for business data
        self.decomposers = {}
        
    def analyze(self, data: np.ndarray, timestamps: np.ndarray) -> Dict[str, Any]:
        """Perform multi-scale temporal analysis"""
        
        results = {
            'patterns': [],
            'anomalies': [],
            'cycles': [],
            'trends': []
        }
        
        # Wavelet transform across all scales
        cwt_matrix = signal.cwt(data, signal.morlet2, self.scales)
        
        # Find patterns at each scale
        for i, scale in enumerate(self.scales):
            scale_data = cwt_matrix[i, :]
            
            # Detect patterns
            patterns = self._detect_patterns_at_scale(scale_data, scale)
            results['patterns'].extend(patterns)
            
            # Detect anomalies
            anomalies = self._detect_anomalies_at_scale(scale_data, scale, timestamps)
            results['anomalies'].extend(anomalies)
        
        # Detect cycles
        results['cycles'] = self._detect_cycles(data, timestamps)
        
        # Detect trends
        results['trends'] = self._detect_trends(data, timestamps)
        
        return results
    
    def _detect_patterns_at_scale(self, scale_data: np.ndarray, scale: float) -> List[Dict]:
        """Detect patterns at a specific time scale"""
        
        patterns = []
        
        # Find peaks
        peaks, properties = signal.find_peaks(
            np.abs(scale_data),
            prominence=np.std(scale_data) * 2
        )
        
        for peak in peaks:
            patterns.append({
                'type': 'peak',
                'scale': scale,
                'position': peak,
                'strength': properties['prominences'][0] if 'prominences' in properties else 0
            })
        
        return patterns
    
    def _detect_anomalies_at_scale(
        self,
        scale_data: np.ndarray,
        scale: float,
        timestamps: np.ndarray
    ) -> List[Dict]:
        """Detect anomalies at a specific scale"""
        
        anomalies = []
        
        # Z-score based anomaly detection
        z_scores = np.abs(stats.zscore(scale_data))
        threshold = 3
        
        anomaly_indices = np.where(z_scores > threshold)[0]
        
        for idx in anomaly_indices:
            anomalies.append({
                'timestamp': timestamps[idx] if idx < len(timestamps) else None,
                'scale': scale,
                'z_score': z_scores[idx],
                'severity': min(z_scores[idx] / threshold, 5.0)
            })
        
        return anomalies
    
    def _detect_cycles(self, data: np.ndarray, timestamps: np.ndarray) -> List[Dict]:
        """Detect cyclical patterns"""
        
        cycles = []
        
        # FFT for frequency analysis
        fft = np.fft.fft(data)
        frequencies = np.fft.fftfreq(len(data))
        
        # Find dominant frequencies
        power = np.abs(fft) ** 2
        dominant_freq_idx = np.argsort(power)[-10:]  # Top 10 frequencies
        
        for idx in dominant_freq_idx:
            if frequencies[idx] > 0:  # Only positive frequencies
                period = 1 / frequencies[idx]
                cycles.append({
                    'period': period,
                    'frequency': frequencies[idx],
                    'power': power[idx],
                    'type': self._classify_cycle(period)
                })
        
        return cycles
    
    def _detect_trends(self, data: np.ndarray, timestamps: np.ndarray) -> List[Dict]:
        """Detect trend patterns"""
        
        trends = []
        
        # Polynomial fitting for trend
        x = np.arange(len(data))
        
        for degree in [1, 2, 3]:  # Linear, quadratic, cubic
            coeffs = np.polyfit(x, data, degree)
            poly = np.poly1d(coeffs)
            
            # Calculate R-squared
            y_pred = poly(x)
            ss_res = np.sum((data - y_pred) ** 2)
            ss_tot = np.sum((data - np.mean(data)) ** 2)
            r_squared = 1 - (ss_res / ss_tot) if ss_tot > 0 else 0
            
            if r_squared > 0.5:  # Significant trend
                trends.append({
                    'type': f'polynomial_{degree}',
                    'coefficients': coeffs.tolist(),
                    'r_squared': r_squared,
                    'direction': 'increasing' if coeffs[0] > 0 else 'decreasing'
                })
        
        return trends
    
    def _classify_cycle(self, period: float) -> str:
        """Classify cycle type based on period"""
        
        if period < 1:
            return 'sub_second'
        elif period < 60:
            return 'seconds'
        elif period < 3600:
            return 'minutes'
        elif period < 86400:
            return 'hourly'
        elif period < 604800:
            return 'daily'
        elif period < 2592000:
            return 'weekly'
        elif period < 31536000:
            return 'monthly'
        else:
            return 'yearly'

@numba.jit(nopython=True)
def fast_correlation(x: np.ndarray, y: np.ndarray, max_lag: int) -> np.ndarray:
    """Fast correlation calculation with Numba"""
    
    correlations = np.zeros(max_lag * 2 + 1)
    
    for lag in range(-max_lag, max_lag + 1):
        if lag < 0:
            corr = np.corrcoef(x[:lag], y[-lag:])[0, 1]
        elif lag > 0:
            corr = np.corrcoef(x[lag:], y[:-lag])[0, 1]
        else:
            corr = np.corrcoef(x, y)[0, 1]
        
        correlations[lag + max_lag] = corr if not np.isnan(corr) else 0
    
    return correlations

    async def _discover_multi_scale_patterns(

    async def _discover_lag_effects(
        self,
        data: Dict[str, pd.DataFrame],
        max_lag_days: int = 3650  # 10 years
    ) -> List[TemporalPattern]:
        """Discover lag effects between variables"""
        
        patterns = []
        
        # Get all numeric columns across all dataframes
        all_series = []
        for name, df in data.items():
            for col in df.select_dtypes(include=[np.number]).columns:
                all_series.append((f"{name}.{col}", df[col].values))
        
        # Test all pairs for lag effects
        for i, (name1, series1) in enumerate(all_series):
            for j, (name2, series2) in enumerate(all_series):
                if i >= j:
                    continue
                
                # Ensure same length
                min_len = min(len(series1), len(series2))
                s1 = series1[:min_len]
                s2 = series2[:min_len]
                
                # Calculate lagged correlations
                max_lag = min(max_lag_days, min_len // 2)
                correlations = fast_correlation(s1, s2, max_lag)
                
                # Find significant lags
                threshold = 0.5
                significant_lags = np.where(np.abs(correlations) > threshold)[0]
                
                for lag_idx in significant_lags:
                    lag = lag_idx - max_lag
                    correlation = correlations[lag_idx]
                    
                    pattern = TemporalPattern(
                        pattern_id=f"LAG_{len(patterns):05d}",
                        time_scale=abs(lag) * 86400,  # Convert to seconds
                        pattern_type='lag_effect',
                        strength=abs(correlation),
                        lag=lag,
                        impact=0,
                        confidence=abs(correlation),
                        evidence={
                            'cause': name1 if lag > 0 else name2,
                            'effect': name2 if lag > 0 else name1,
                            'lag_days': abs(lag),
                            'correlation': correlation
                        }
                    )
                    patterns.append(pattern)
        
        return patterns
    
    async def _discover_causal_chains(
        self,
        data: Dict[str, pd.DataFrame]
    ) -> List[TemporalPattern]:
        """Discover causal chains using Granger causality"""
        
        patterns = []
        
        # Prepare data for Granger causality test
        combined_data = []
        column_names = []
        
        for name, df in data.items():
            for col in df.select_dtypes(include=[np.number]).columns:
                if len(df) > 100:  # Need sufficient data
                    combined_data.append(df[col].values[:1000])  # Limit size
                    column_names.append(f"{name}.{col}")
        
        if len(combined_data) < 2:
            return patterns
        
        # Stack data
        data_matrix = np.column_stack(combined_data)
        
        # Test Granger causality for all pairs
        for i in range(len(column_names)):
            for j in range(len(column_names)):
                if i == j:
                    continue
                
                try:
                    # Simplified Granger test
                    test_data = data_matrix[:, [i, j]]
                    
                    # Use statsmodels Granger causality test
                    max_lag = min(10, len(test_data) // 10)
                    
                    # ADF test for stationarity
                    adf_i = adfuller(test_data[:, 0])[1]
                    adf_j = adfuller(test_data[:, 1])[1]
                    
                    if adf_i < 0.05 and adf_j < 0.05:  # Both stationary
                        # Granger test
                        result = grangercausalitytests(test_data, max_lag, verbose=False)
                        
                        # Check p-values
                        p_values = [result[lag][0]['ssr_ftest'][1] for lag in range(1, max_lag + 1)]
                        min_p = min(p_values)
                        best_lag = p_values.index(min_p) + 1
                        
                        if min_p < 0.05:  # Significant causality
                            pattern = TemporalPattern(
                                pattern_id=f"CAUSAL_{len(patterns):05d}",
                                time_scale=best_lag * 86400,  # Assuming daily data
                                pattern_type='causal_chain',
                                strength=1 - min_p,
                                lag=best_lag,
                                impact=0,
                                confidence=1 - min_p,
                                evidence={
                                    'cause': column_names[i],
                                    'effect': column_names[j],
                                    'lag': best_lag,
                                    'p_value': min_p
                                }
                            )
                            patterns.append(pattern)
                            
                except Exception as e:
                    continue
        
        return patterns
    
    async def _predict_anomalies(

class AnomalyPredictor(nn.Module):
    """LSTM-based anomaly predictor"""
    
    def __init__(self, input_dim: int = 1, hidden_dim: int = 64, num_layers: int = 2):
        super().__init__()
        
        self.lstm = nn.LSTM(
            input_size=input_dim,
            hidden_size=hidden_dim,
            num_layers=num_layers,
            batch_first=True
        )
        
        self.decoder = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, 1)
        )
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        lstm_out, _ = self.lstm(x)
        predictions = self.decoder(lstm_out)
        return predictions
    
    def predict(self, data: np.ndarray, future_steps: int = 30) -> List[Dict]:
        """Predict future anomalies"""
        
        # Simplified prediction
        anomalies = []
        
        # Statistical approach for demo
        mean = np.mean(data)
        std = np.std(data)
        
        for i in range(future_steps):
            # Random anomaly generation for demo
            if np.random.random() > 0.95:  # 5% chance
                anomalies.append({
                    'time_until': i,
                    'severity': np.random.uniform(3, 5),
                    'confidence': np.random.uniform(0.6, 0.95),
                    'expected_value': mean + np.random.randn() * std * 3
                })
        
        return anomalies

@dataclass
class AlternativeTimeline:

@dataclass
class Decision:
    """Represents a business decision"""
    decision_id: str
    timestamp: datetime
    type: str
    parameters: Dict[str, Any]
    actual_outcome: Optional[float]
    alternatives: List[Dict[str, Any]]

class BusinessSimulator:
    """Simulates business outcomes under different decisions"""
    
    def __init__(self):
        self.state_transition_model = self._build_state_transition_model()
        self.outcome_predictor = self._build_outcome_predictor()
        
    def _build_state_transition_model(self) -> nn.Module:
        """Build neural state transition model"""
        
        class StateTransition(nn.Module):
            def __init__(self, state_dim: int = 256, action_dim: int = 128):
                super().__init__()
                
                self.transition = nn.Sequential(
                    nn.Linear(state_dim + action_dim, 512),
                    nn.ReLU(),
                    nn.Dropout(0.1),
                    nn.Linear(512, 512),
                    nn.ReLU(),
                    nn.Dropout(0.1),
                    nn.Linear(512, state_dim)
                )
                
                # Uncertainty estimation
                self.uncertainty = nn.Sequential(
                    nn.Linear(state_dim + action_dim, 256),
                    nn.ReLU(),
                    nn.Linear(256, state_dim)
                )
            
            def forward(self, state: torch.Tensor, action: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
                combined = torch.cat([state, action], dim=-1)
                next_state = self.transition(combined)
                uncertainty = torch.sigmoid(self.uncertainty(combined))
                return next_state, uncertainty
        
        return StateTransition()
    
    def _build_outcome_predictor(self) -> nn.Module:
        """Build outcome prediction model"""
        
        class OutcomePredictor(nn.Module):
            def __init__(self, state_dim: int = 256):
                super().__init__()
                
                self.predictor = nn.Sequential(
                    nn.Linear(state_dim, 256),
                    nn.ReLU(),
                    nn.Dropout(0.1),
                    nn.Linear(256, 128),
                    nn.ReLU(),
                    nn.Linear(128, 5)  # Revenue, Cost, Profit, Growth, Risk
                )
            
            def forward(self, state: torch.Tensor) -> torch.Tensor:
                return self.predictor(state)
        
        return OutcomePredictor()
    
    def simulate_timeline(
        self,
        initial_state: np.ndarray,
        decisions: List[Decision],
        horizon: int = 365
    ) -> Dict[str, Any]:
        """Simulate a business timeline"""
        
        state = torch.tensor(initial_state, dtype=torch.float32)
        trajectory = [state.numpy()]
        outcomes = []
        
        for day in range(horizon):
            # Get decisions for this day
            daily_decisions = [d for d in decisions if d.timestamp.day == day]
            
            if daily_decisions:
                # Aggregate decisions into action
                action = self._aggregate_decisions(daily_decisions)
            else:
                # No-op action
                action = torch.zeros(128)
            
            # State transition
            with torch.no_grad():
                next_state, uncertainty = self.state_transition_model(
                    state.unsqueeze(0),
                    action.unsqueeze(0)
                )
                
                # Add noise based on uncertainty
                noise = torch.randn_like(next_state) * uncertainty
                next_state = next_state + noise
                
                # Predict outcomes
                outcome = self.outcome_predictor(next_state)
                
                state = next_state.squeeze(0)
                trajectory.append(state.numpy())
                outcomes.append(outcome.squeeze(0).numpy())
        
        return {
            'trajectory': np.array(trajectory),
            'outcomes': np.array(outcomes),
            'final_state': state.numpy()
        }
    
    def _aggregate_decisions(self, decisions: List[Decision]) -> torch.Tensor:
        """Aggregate multiple decisions into action vector"""
        
        action = torch.zeros(128)
        
        for decision in decisions:
            # Encode decision (simplified)
            decision_encoding = self._encode_decision(decision)
            action += decision_encoding
        
        return torch.tanh(action)  # Normalize
    
    def _encode_decision(self, decision: Decision) -> torch.Tensor:
        """Encode decision to vector"""
        
        # Simplified encoding
        encoding = torch.randn(128) * 0.1
        
        # Encode decision type
        type_mapping = {
            'pricing': 0,
            'marketing': 1,
            'operations': 2,
            'product': 3,
            'hr': 4
        }
        
        if decision.type in type_mapping:
            encoding[type_mapping[decision.type] * 20:(type_mapping[decision.type] + 1) * 20] = 1.0
        
        return encoding

class CounterfactualRealityEngine:

    def _identify_decision_points(
        self,
        historical_decisions: List[Decision]
    ) -> List[Decision]:
        """Identify critical decision points"""
        
        critical_decisions = []
        
        for decision in historical_decisions:
            # Calculate decision impact
            impact = self._estimate_decision_impact(decision)
            
            if impact > 100000:  # Significant impact
                critical_decisions.append(decision)
                
                # Add to decision graph
                self.decision_graph.add_node(
                    decision.decision_id,
                    decision=decision,
                    impact=impact
                )
        
        # Find decision dependencies
        for i, d1 in enumerate(critical_decisions):
            for j, d2 in enumerate(critical_decisions):

                    self.decision_graph.add_edge(d1.decision_id, d2.decision_id)
        
        return critical_decisions
    
    def _estimate_decision_impact(self, decision: Decision) -> float:
        """Estimate the impact of a decision"""
        
        if decision.actual_outcome:
            return abs(decision.actual_outcome)
        
        # Estimate based on decision type
        impact_estimates = {
            'pricing': 1000000,
            'marketing': 500000,
            'operations': 750000,
            'product': 2000000,
            'hr': 250000,
            'strategic': 5000000
        }
        
        return impact_estimates.get(decision.type, 100000)
    
    def _decisions_related(self, d1: Decision, d2: Decision) -> bool:
        """Check if two decisions are related"""
        
        # Time proximity
        time_diff = abs((d1.timestamp - d2.timestamp).days)
        if time_diff < 30:
            return True
        
        # Same type
        if d1.type == d2.type:
            return True
        
        # Parameter overlap
        params1 = set(d1.parameters.keys())
        params2 = set(d2.parameters.keys())
        if len(params1.intersection(params2)) > 0:
            return True
        
        return False
    
    def _generate_alternatives(
        self,
        decision_points: List[Decision],
        num_alternatives: int
    ) -> List[List[Decision]]:
        """Generate alternative decision combinations"""
        
        alternatives = []
        
        for _ in range(num_alternatives):
            alternative_decisions = []
            
            for decision in decision_points:
                # Generate alternative for this decision
                alt_decision = self._mutate_decision(decision)
                alternative_decisions.append(alt_decision)
            
            alternatives.append(alternative_decisions)
        
        # Add some optimized alternatives using genetic algorithm
        optimized = self._genetic_optimization(decision_points, 100)
        alternatives.extend(optimized)
        
        return alternatives
    
    def _mutate_decision(self, decision: Decision) -> Decision:
        """Create alternative version of a decision"""
        
        mutated = copy.deepcopy(decision)
        mutated.decision_id = f"{decision.decision_id}_alt_{np.random.randint(10000)}"
        
        # Mutate parameters
        for param, value in mutated.parameters.items():
            if isinstance(value, (int, float)):
                # Numerical mutation
                mutated.parameters[param] = value * np.random.uniform(0.5, 1.5)
            elif isinstance(value, str):
                # Categorical mutation
                alternatives = decision.alternatives if decision.alternatives else [value]
                if alternatives:
                    mutated.parameters[param] = np.random.choice(alternatives)
            elif isinstance(value, bool):
                # Boolean flip
                mutated.parameters[param] = not value if np.random.random() > 0.5 else value
        
        return mutated
    
    def _genetic_optimization(
        self,
        decision_points: List[Decision],
        num_generations: int
    ) -> List[List[Decision]]:
        """Use genetic algorithm to find optimal decision combinations"""
        
        def fitness_function(individual):
            # Simulate and evaluate
            decisions = [
                self._decode_gene(gene, decision)
                for gene, decision in zip(individual, decision_points)
            ]
            
            # Quick evaluation
            score = sum(self._estimate_decision_impact(d) for d in decisions)
            return -score  # Minimize negative score
        
        # Define bounds for each decision parameter
        bounds = [(0, 1) for _ in decision_points]
        
        # Run differential evolution
        result = differential_evolution(
            fitness_function,
            bounds,
            maxiter=num_generations,
            popsize=15,
            workers=1
        )
        
        # Generate multiple solutions from the optimization
        solutions = []
        for _ in range(10):
            # Add noise to optimal solution
            noisy_solution = result.x + np.random.randn(len(result.x)) * 0.1
            noisy_solution = np.clip(noisy_solution, 0, 1)
            
            decisions = [
                self._decode_gene(gene, decision)
                for gene, decision in zip(noisy_solution, decision_points)
            ]
            
            solutions.append(decisions)
        
        return solutions
    
    def _decode_gene(self, gene: float, decision: Decision) -> Decision:
        """Decode genetic algorithm gene to decision"""
        
        decoded = copy.deepcopy(decision)
        
        # Use gene value to modify decision parameters
        for param, value in decoded.parameters.items():
            if isinstance(value, (int, float)):
                decoded.parameters[param] = value * (0.5 + gene)
        
        return decoded
    
    async def _simulate_timelines(
        self,
        initial_state: Dict[str, Any],
        alternatives: List[List[Decision]]
    ) -> List[AlternativeTimeline]:
        """Simulate alternative timelines in parallel"""
        
        timelines = []
        
        # Convert initial state to vector
        state_vector = self._encode_state(initial_state)
        
        # Batch simulation for efficiency
        batch_size = 100
        for i in range(0, len(alternatives), batch_size):
            batch = alternatives[i:i+batch_size]
            
            batch_results = []
            for alt_decisions in batch:
                # Simulate timeline
                result = self.simulator.simulate_timeline(
                    state_vector,
                    alt_decisions,
                    horizon=365
                )
                
                # Create timeline object
                timeline = AlternativeTimeline(
                    timeline_id=f"TL_{i:07d}",
                    divergence_point=alt_decisions[0].timestamp if alt_decisions else datetime.now(),
                    changes_made=[self._decision_to_dict(d) for d in alt_decisions],
                    outcomes=self._extract_outcomes(result),
                    probability=self._calculate_probability(alt_decisions),
                    improvement=0  # Will be calculated
                )
                
                batch_results.append(timeline)
            
            timelines.extend(batch_results)
        
        return timelines
    
    def _encode_state(self, state: Dict[str, Any]) -> np.ndarray:
        """Encode business state to vector"""
        
        # Simplified encoding
        vector = []
        
        for key, value in state.items():
            if isinstance(value, (int, float)):
                vector.append(float(value))
            elif isinstance(value, list) and value and isinstance(value[0], (int, float)):
                vector.extend([float(v) for v in value[:10]])
        
        # Pad to fixed size
        target_size = 256
        if len(vector) < target_size:
            vector.extend([0.0] * (target_size - len(vector)))
        else:
            vector = vector[:target_size]
        
        return np.array(vector)
    
    def _decision_to_dict(self, decision: Decision) -> Dict[str, Any]:
        """Convert decision to dictionary"""
        
        return {
            'id': decision.decision_id,
            'type': decision.type,
            'timestamp': decision.timestamp.isoformat(),
            'parameters': decision.parameters
        }
    
    def _extract_outcomes(self, simulation_result: Dict[str, Any]) -> Dict[str, float]:
        """Extract outcomes from simulation"""
        
        outcomes = simulation_result['outcomes']
        
        if len(outcomes) > 0:
            # Average outcomes over timeline
            avg_outcomes = np.mean(outcomes, axis=0)
            
            return {
                'revenue': avg_outcomes[0] * 1000000,
                'cost': avg_outcomes[1] * 1000000,
                'profit': avg_outcomes[2] * 1000000,
                'growth': avg_outcomes[3],
                'risk': avg_outcomes[4]
            }
        
        return {
            'revenue': 0,
            'cost': 0,
            'profit': 0,
            'growth': 0,
            'risk': 0
        }
    
    def _calculate_probability(self, decisions: List[Decision]) -> float:
        """Calculate probability of timeline occurring"""
        
        if not decisions:
            return 1.0
        
        # Product of individual decision probabilities
        probability = 1.0
        
        for decision in decisions:
            # Estimate probability based on how different from original
            if hasattr(decision, 'original'):
                similarity = self._calculate_decision_similarity(decision, decision.original)
                probability *= similarity
            else:
                probability *= 0.5
        
        return max(probability, 0.01)
    
    def _calculate_decision_similarity(self, d1: Decision, d2: Decision) -> float:
        """Calculate similarity between decisions"""
        
        if d1.type != d2.type:
            return 0.1
        
        # Compare parameters
        similarity = 0.0
        param_count = 0
        
        for param in d1.parameters:
            if param in d2.parameters:
                v1 = d1.parameters[param]
                v2 = d2.parameters[param]
                
                if isinstance(v1, (int, float)) and isinstance(v2, (int, float)):
                    # Numerical similarity
                    diff = abs(v1 - v2) / max(abs(v1), abs(v2), 1)
                    similarity += 1 - min(diff, 1)
                elif v1 == v2:
                    similarity += 1
                
                param_count += 1
        
        return similarity / max(param_count, 1)
    
    def _evaluate_timelines(
        self,
        timelines: List[AlternativeTimeline],
        current_state: Dict[str, Any]
    ) -> List[AlternativeTimeline]:
        """Evaluate and rank timelines"""
        
        # Calculate current baseline
        baseline_profit = current_state.get('annual_profit', 0)
        
        for timeline in timelines:
            # Calculate improvement
            timeline.improvement = (
                (timeline.outcomes['profit'] - baseline_profit) /
                max(baseline_profit, 1)
            )
        
        # Sort by improvement
        timelines.sort(key=lambda t: t.improvement, reverse=True)
        
        return timelines
    
    def _find_optimal_timeline(
        self,
        timelines: List[AlternativeTimeline]
    ) -> AlternativeTimeline:
        """Find the optimal timeline considering risk and probability"""
        
        best_score = -float('inf')
        best_timeline = timelines[0] if timelines else None
        
        for timeline in timelines:
            # Score = improvement * probability / (1 + risk)
            score = (
                timeline.improvement *
                timeline.probability /
                (1 + timeline.outcomes.get('risk', 0))
            )
            
            if score > best_score:
                best_score = score
                best_timeline = timeline
        
        return best_timeline
    
    def compute_timeline_delta(
        self,
        current: Dict[str, Any],
        optimal: AlternativeTimeline
    ) -> Dict[str, Any]:
        """Compute exact changes needed to reach optimal timeline"""
        
        delta = {
            'decisions_to_change': optimal.changes_made,
            'expected_improvement': optimal.improvement,
            'probability_of_success': optimal.probability,
            'implementation_steps': self._generate_implementation_plan(optimal),
            'risks': self._identify_risks(optimal),
            'timeline': self._create_timeline(optimal)
        }
        
        return delta
    
    def _generate_implementation_plan(
        self,
        timeline: AlternativeTimeline
    ) -> List[Dict[str, Any]]:
        """Generate step-by-step implementation plan"""
        
        steps = []
        
        for i, change in enumerate(timeline.changes_made):
            steps.append({
                'step': i + 1,
                'action': f"Modify {change['type']} decision",
                'details': change['parameters'],
                'timing': change['timestamp'],
                'dependencies': self._find_dependencies(change, timeline.changes_made),
                'expected_impact': timeline.outcomes['profit'] / len(timeline.changes_made)
            })
        
        return steps
    
    def _find_dependencies(
        self,
        change: Dict[str, Any],
        all_changes: List[Dict[str, Any]]
    ) -> List[int]:
        """Find dependencies for a change"""
        
        dependencies = []
        
        for i, other_change in enumerate(all_changes):
            if other_change == change:
                continue
            
            # Check if this change depends on the other
            if self._change_depends_on(change, other_change):
                dependencies.append(i)
        
        return dependencies
    
    def _change_depends_on(self, change1: Dict, change2: Dict) -> bool:
        """Check if change1 depends on change2"""
        
        # Time dependency
        t1 = datetime.fromisoformat(change1['timestamp'])
        t2 = datetime.fromisoformat(change2['timestamp'])
        
        if t1 > t2:
            # Check parameter dependencies
            for param in change1['parameters']:
                if param in change2['parameters']:
                    return True
        
        return False
    
    def _identify_risks(
        self,
        timeline: AlternativeTimeline
    ) -> List[Dict[str, Any]]:
        """Identify risks in the alternative timeline"""
        
        risks = []
        
        # High change risk
        if len(timeline.changes_made) > 10:
            risks.append({
                'type': 'implementation_complexity',
                'severity': 'high',
                'mitigation': 'Phase implementation over time'
            })
        
        # Low probability risk
        if timeline.probability < 0.3:
            risks.append({
                'type': 'low_success_probability',
                'severity': 'medium',
                'mitigation': 'Prepare contingency plans'
            })
        
        # High risk score
        if timeline.outcomes.get('risk', 0) > 0.7:
            risks.append({
                'type': 'high_business_risk',
                'severity': 'high',
                'mitigation': 'Implement risk management framework'
            })
        
        return risks
    
    def _create_timeline(
        self,
        timeline: AlternativeTimeline
    ) -> List[Dict[str, Any]]:
        """Create implementation timeline"""
        
        events = []
        
        start_date = datetime.now()
        
        for i, change in enumerate(timeline.changes_made):
            events.append({
                'date': (start_date + timedelta(days=i*7)).isoformat(),
                'event': f"Implement {change['type']} change",
                'milestone': f"Milestone {i+1}",
                'expected_progress': (i + 1) / len(timeline.changes_made)
            })
        
        return events

                }
            else:
                return {
                    'status': 'blocked',
                    'reason': violations
                }
        
        # 3. Execute plan
        print(f"✅ Safety checks passed. Executing plan...")
        result = await self._execute_plan(plan)
        
        # 4. Verify results
        if result['success']:
            print(f"🎯 Execution successful! Impact: ${result['actual_impact']:,.0f}")
        else:
            print(f"❌ Execution failed. Rolling back...")
            await self.safety_checker.rollback_manager.execute_rollback(
                plan.rollback_plan,
                self.connector
            )
        
        # 5. Record execution
        self.execution_history.append({
            'discovery_id': discovery.id,
            'plan': plan,
            'result': result,
            'timestamp': datetime.now()
        })
        
        return result
    
    async def _generate_execution_plan(self, discovery: Any) -> ExecutionPlan:
        """Generate execution plan from discovery"""
        
        steps = []
        
        # Determine execution type
        if 'database' in discovery.evidence:
            # Database execution
            sql = self.code_generator.generate_execution_code(discovery, 'sql')
            steps.append({
                'type': 'database_update',
                'sql': sql,
                'savepoint': f"sp_{discovery.id}"
            })
            
        if 'api' in discovery.evidence:
            # API execution
            api_calls = self.code_generator.generate_execution_code(discovery, 'api')
            steps.append({
                'type': 'api_call',
                'calls': json.loads(api_calls)
            })
        
        # Generate rollback plan
        rollback_plan = self.safety_checker.rollback_manager.create_rollback_plan(steps)
        
        # Define safety checks
        safety_checks = [
            lambda: self._check_system_health(),
            lambda: self._check_dependencies(),
            lambda: self._check_resource_availability()
        ]
        
        plan = ExecutionPlan(
            plan_id=f"PLAN_{discovery.id}",
            discovery=discovery,
            steps=steps,
            rollback_plan=rollback_plan,
            safety_checks=safety_checks,
            estimated_impact=discovery.impact,
            confidence=discovery.confidence
        )
        
        return plan
    
    async def _execute_plan(self, plan: ExecutionPlan) -> Dict[str, Any]:
        """Execute the plan"""
        
        results = []
        
        try:
            for step in plan.steps:
                print(f"  → Executing step: {step['type']}")
                
                if step['type'] == 'database_update':
                    result = await self.connector.execute_sql(step['sql'])
                elif step['type'] == 'api_call':
                    result = await self._execute_api_calls(step['calls'])
                else:

                results.append(result)
            
            # Calculate actual impact

            return {
                'success': True,
                'results': results,
                'actual_impact': actual_impact,
                'execution_time': datetime.now()
            }
            
        except Exception as e:
            return {
                'success': False,
                'error': str(e),
                'partial_results': results
            }
    
    async def _execute_api_calls(self, calls: Dict[str, Any]) -> List[Any]:
